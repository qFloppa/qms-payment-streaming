### Task 1: Native streaming contract and executable test harness

**Files:** `contracts/RillStream.sol`, `contracts/test/Receivers.sol`, `scripts/compile-contract.mjs`, `scripts/local-chain.mjs`, `scripts/deploy.mjs`, `tests/contracts.test.mjs`, `src/contracts/rill.ts`, `src/contracts/deployment.ts`, `package.json`, `package-lock.json`.

**Interfaces:** Export `rillAbi`, `rillRuntimeHash` from `src/contracts/rill.ts`; export `rillBytecode` from `src/contracts/deployment.ts`. The generated ABI must reflect these stored fields verbatim:

```solidity
struct Stream {
  address sender;
  address recipient;
  uint256 deposit;
  uint256 withdrawn;
  uint256 refunded;
  uint64 startTime;
  uint64 endTime;
  uint64 cancelledAt;
}
function createStream(address recipient, uint64 duration) external payable returns (uint256);
function getStream(uint256 streamId) external view returns (Stream memory);
function balances(uint256 streamId) external view returns (uint256 unlocked, uint256 available, uint256 scheduled);
function streamCount(address wallet) external view returns (uint256);
function streamsFor(address wallet,uint256 cursor,uint256 limit) external view returns (uint256[] memory ids,uint256 nextCursor);
function withdraw(uint256 streamId) external;
function cancel(uint256 streamId) external;
```
